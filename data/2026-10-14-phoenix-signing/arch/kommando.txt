# The WS signing image, as the release job builds it, 06:27:
DOCKER_CONFIG=/private/tmp/rb1-docker docker buildx build --builder default --load -t ws-signing:rb-test $W/deployment/signing

# A throwaway test CA and code-signing certificate, 06:30, in $O/keys (the
# private keys and the keystore's password stay there):
openssl rand -hex 16 > password.txt
openssl req -x509 -newkey rsa:3072 -sha256 -days 3 -nodes -subj "/CN=RB1 throwaway test CA (thesis, 14 Oct 2026)" -keyout ca.key -out ca.pem -addext "basicConstraints=critical,CA:TRUE" -addext "keyUsage=critical,keyCertSign,cRLSign"
openssl req -newkey rsa:3072 -sha256 -nodes -subj "/CN=RB1 throwaway test code signing (thesis, 14 Oct 2026)" -keyout cs.key -out cs.csr
printf 'basicConstraints=CA:FALSE\nkeyUsage=critical,digitalSignature\nextendedKeyUsage=codeSigning\n' > cs.ext
openssl x509 -req -in cs.csr -CA ca.pem -CAkey ca.key -CAcreateserial -days 3 -sha256 -extfile cs.ext -out cs.pem
SIGN_KEYSTORE_PASSWORD=$(cat password.txt) openssl pkcs12 -export -inkey cs.key -in cs.pem -certfile ca.pem -name test -out test.p12 -passout env:SIGN_KEYSTORE_PASSWORD
cat ca.pem $W/deployment/signing/microsoft-identity-verification-root-2020.pem > verify.pem

# The probe, 06:30: A and B are the first builds of W2-none and W2-all.
