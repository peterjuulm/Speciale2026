probe() { local tag=$1 ns=$2 clihome=$3; mkdir -p $P/home-$tag; rm -rf $P/proj/.local $P/proj/obj
  local e=(env -i HOME=$P/home-$tag PATH=$DN:/usr/bin DOTNET_ROOT=$DN DOTNET_NOLOGO=1 DOTNET_CLI_TELEMETRY_OPTOUT=1 MSBUILDDISABLENODEREUSE=1 $clihome)
  if [ "$ns" = ns ]; then (cd $P/proj && unshare -rnp --fork "${e[@]}" dotnet restore -v q > /dev/null 2>&1); else (cd $P/proj && "${e[@]}" dotnet restore -v q > /dev/null 2>&1); fi
  echo "$tag: exit $?; in cwd: $([ -d $P/proj/.local ] && echo yes || echo no); in HOME: $(find $P/home-$tag -path '*NuGet/Migrations*' | head -1 | sed "s|$P/||")"; }
probe a ns DOTNET_CLI_HOME=$P/clihome-a
probe b nons DOTNET_CLI_HOME=$P/clihome-b
probe c ns ""
probe d nons ""
find $P -path '*Migrations' -type d | sed "s|$P/||"
